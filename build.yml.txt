name: Build Litematica AutoBuild

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Extract source
        run: |
          if [ -f source.zip ]; then
            unzip -oq source.zip
          fi
          # Nếu bị bọc folder thì đẩy lên
          if [ -d source ] && [ -f source/build.gradle ]; then
            mv source/* .
            rmdir source 2>/dev/null || true
          fi
          echo "=== Java files ==="
          find src -name "*.java" | head -20 || echo "No java files"

      - uses: actions/setup-java@v4
        with:
          java-version: '21'
          distribution: 'temurin'

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '9.2.1'

      - name: Build
        run: gradle build --no-daemon

      - uses: actions/upload-artifact@v4
        with:
          name: mod-jar
          path: build/libs/*.jar
          if-no-files-found: error
